<script>
const ARTIST_ID = 1734510859; // Lety Videz en Apple Music / iTunes
const app = document.getElementById('app');
const audio = document.getElementById('audio');
let albums = [];

// JSONP para evitar problemas de CORS con la API de iTunes
function itunes(url) {
  return new Promise((resolve, reject) => {
    const cb = 'cb' + Math.random().toString(36).slice(2);
    const s = document.createElement('script');
    window[cb] = d => { resolve(d); delete window[cb]; s.remove(); };
    s.onerror = reject;
    s.src = url + '&callback=' + cb;
    document.body.appendChild(s);
  });
}
const big = url => url.replace('100x100bb', '600x600bb');
const mmss = ms => { const s = Math.round(ms/1000); return Math.floor(s/60) + ':' + String(s%60).padStart(2,'0'); };
const esc = s => s.replace(/[&<>"]/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));

async function loadAlbums() {
  const d = await itunes(`https://itunes.apple.com/lookup?id=${ARTIST_ID}&entity=album&limit=200&country=es`);
  albums = d.results.filter(r => r.wrapperType === 'collection')
    .sort((a,b) => b.releaseDate.localeCompare(a.releaseDate));
}

function showList() {
  app.innerHTML = `<div class="grid">${albums.map(a => `
    <a class="album" href="#disco/${a.collectionId}">
      <img src="${big(a.artworkUrl100)}" alt="">
      <div><b>${esc(a.collectionName)}</b><small>${a.releaseDate.slice(0,4)} · ${a.trackCount} canciones</small></div>
    </a>`).join('')}</div>`;
}

async function showAlbum(id) {
  app.innerHTML = 'Cargando…';
  const d = await itunes(`https://itunes.apple.com/lookup?id=${id}&entity=song&country=es`);
  const album = d.results.find(r => r.wrapperType === 'collection');
  const songs = d.results.filter(r => r.wrapperType === 'track').sort((a,b) => a.discNumber-b.discNumber || a.trackNumber-b.trackNumber);
  app.innerHTML = `
    <a class="back" href="#">← Todos los discos</a>
    <div class="detail">
      <img src="${big(album.artworkUrl100)}" alt="Carátula">
      <div><h2>${esc(album.collectionName)}</h2><p>${album.releaseDate.slice(0,4)} · ${songs.length} canciones</p></div>
    </div>
    <ol>${songs.map((s,i) => `
      <li data-i="${i}"><span class="n">${s.trackNumber}</span><span class="t">${esc(s.trackName)}</span>
      <span class="d">${mmss(s.trackTimeMillis)}</span>
      <a href="${s.trackViewUrl}" target="_blank" onclick="event.stopPropagation()">completa ↗</a></li>`).join('')}</ol>
    <p class="note">Se reproducen avances de 30 s. Pulsa “completa ↗” para escucharla entera en Apple Music.</p>`;
  app.querySelectorAll('li').forEach(li => li.onclick = () => play(songs, +li.dataset.i));
}

function play(songs, i) {
  const s = songs[i];
  if (!s) return;
  document.querySelectorAll('li').forEach(li => li.classList.toggle('playing', +li.dataset.i === i));
  document.getElementById('now').textContent = '▶ ' + s.trackName;
  document.getElementById('player').classList.add('on');
  audio.src = s.previewUrl;
  audio.play();
  audio.onended = () => play(songs, i + 1); // pasa a la siguiente
}

async function route() {
  if (!albums.length) await loadAlbums();
  const m = location.hash.match(/^#disco\/(\d+)/);
  m ? showAlbum(m[1]) : showList();
  scrollTo(0, 0);
}
addEventListener('hashchange', route);
route().catch(() => app.textContent = 'No se pudo cargar la discografía. Revisa tu conexión.');
</script>
